#Method: Get hint from AI
def get_game_hint(game_state: str) -> str:
    """
    :param game_state: Description of the current game situation.
    :return: Hint text from AI.
    """
    try:
        response = openai.ChatCompletion.create(
            model="gpt-4o-mini",  
            messages=[
                {"role": "system", "content": "You are a helpful hint bot for a game. Give short, useful hints without spoilers."},
                {"role": "user", "content": f"Game situation: {game_state}"}
            ],
            max_tokens=10,  # Limit hint length
            temperature=0.7  # Slight creativity
        )
        return response.choices[0].message["content"].strip()
    except Exception as e:
        return f"Error getting hint: {e}"
    
#Method: Send hint to Arduino using serial
def send_to_arduino(message):
    try:
        with serial.Serial(SERIAL_PORT, BAUD_RATE, timeout=2) as arduino:
            time.sleep(2)  # Wait for Arduino to reset
            arduino.write(message.encode('utf-8'))
            print(f"Sent to Arduino: {message}")
    except serial.SerialException as e:
        print(f"Serial error: {e}")
    except Exception as e:
        print(f"Unexpected error: {e}")
